    await getPool();

const express = require("express");
const router = express.Router();
const sql = require("mssql");
const { getPool } = require("../db");

// =====================================================
// ONLY ADMIN / MANAGER HAVE FULL ACCESS
// =====================================================

const FULL_ACCESS_ROLES = [
  "admin",
  "manager",
];

// =====================================================
// NORMALIZE ROLE
// =====================================================

const normalizeRole = (value) => {
  return String(value || "")
    .trim()
    .toLowerCase()
    .replace(/\s+/g, " ");
};

// =====================================================
// DEFAULT DATE WINDOW
// -----------------------------------------------------
// Applied to /purchases whenever the caller doesn't supply
// fromDate/toDate, so a plain dashboard/list load can't turn
// into an unbounded scan of the whole purchase history.
// =====================================================

const DEFAULT_LOOPBACK_DAYS = 31;

function getDefaultFromDate() {
  const d = new Date();
  d.setDate(d.getDate() - DEFAULT_LOOPBACK_DAYS);
  return d.toISOString().slice(0, 10); // YYYY-MM-DD
}

router.get("/purchases", async (req, res) => {
  try {
    const {
      userCode,
      companyCode,
      memberNumber,
      fromDate,
      toDate,
    } = req.query;

    // =================================================
    // VALIDATE USER CODE
    // =================================================

    if (!userCode || String(userCode).trim() === "") {
      return res.status(400).json({
        success: false,
        message: "userCode is required.",
      });
    }

    const pool = await getPool();

    // =================================================
    // GET LOGGED-IN USER
    // =================================================

    const userResult = await pool
      .request()
      .input("UserCode", sql.Int, Number(userCode))
      .query(`
        SELECT
          u.UserCode,
          u.UserTypeCode,
          u.CompanyCode,
          u.MemberNumber,
          u.UserName,

          t.UserTypeName,

          RTRIM(
            ISNULL(c.Header1, '') +
            ISNULL(c.Header2, '')
          ) AS CompanyName

        FROM tbl_User u

        INNER JOIN tbl_UserType t
          ON u.UserTypeCode = t.UserTypeCode

        OUTER APPLY
        (
          SELECT TOP 1
            co.Header1,
            co.Header2
          FROM tbl_Company co
          WHERE co.CompanyCode = u.CompanyCode
          ORDER BY co.EDNO
        ) c

        WHERE u.UserCode = @UserCode
      `);

    // =================================================
    // USER NOT FOUND
    // =================================================

    if (userResult.recordset.length === 0) {
      return res.status(404).json({
        success: false,
        message: "User not found.",
      });
    }

    const me = userResult.recordset[0];

    // =================================================
    // ROLE
    // =================================================

    const role = normalizeRole(me.UserTypeName);

    const isSecretary =
      Number(me.UserTypeCode) === 2 ||
      role === "secretary" ||
      role === "secretory";

    const isFullAccess =
      !isSecretary &&
      FULL_ACCESS_ROLES.includes(role);

    // =================================================
    // BUILD WHERE
    // =================================================

    const request = pool.request();

    let where = "WHERE 1 = 1";

    // =================================================
    // ADMIN / MANAGER
    // =================================================

    if (isFullAccess) {
      if (
        companyCode &&
        String(companyCode).trim() !== ""
      ) {
        request.input(
          "FilterCompanyCode",
          sql.Int,
          Number(companyCode)
        );

        where += `
          AND p.CompanyCode = @FilterCompanyCode
        `;
      }

      if (
        memberNumber &&
        String(memberNumber).trim() !== ""
      ) {
        request.input(
          "FilterMemberNumber",
          sql.VarChar(50),
          String(memberNumber).trim()
        );

        where += `
          AND CONVERT(varchar(50), p.MemberCode)
              = @FilterMemberNumber
        `;
      }
    }

    // =================================================
    // SECRETARY
    // =================================================

    else if (isSecretary) {
      if (
        me.CompanyCode === null ||
        me.CompanyCode === undefined
      ) {
        return res.status(400).json({
          success: false,
          message: "Secretary has no registered society.",
        });
      }

      request.input(
        "SecretaryCompanyCode",
        sql.Int,
        Number(me.CompanyCode)
      );

      where += `
        AND p.CompanyCode = @SecretaryCompanyCode
      `;

      if (
        memberNumber &&
        String(memberNumber).trim() !== ""
      ) {
        request.input(
          "SecretaryMemberNumber",
          sql.VarChar(50),
          String(memberNumber).trim()
        );

        where += `
          AND CONVERT(varchar(50), p.MemberCode)
              = @SecretaryMemberNumber
        `;
      }
    }

    // =================================================
    // NORMAL MEMBER
    // =================================================

    else {
      if (
        me.CompanyCode === null ||
        me.CompanyCode === undefined
      ) {
        return res.status(400).json({
          success: false,
          message: "User has no registered society.",
        });
      }

      request.input(
        "UserCompanyCode",
        sql.Int,
        Number(me.CompanyCode)
      );

      request.input(
        "UserMemberNumber",
        sql.VarChar(50),
        String(me.MemberNumber || "").trim()
      );

      where += `
        AND p.CompanyCode = @UserCompanyCode

        AND CONVERT(varchar(50), p.MemberCode)
            = @UserMemberNumber
      `;
    }

    // =================================================
    // FROM DATE / TO DATE
    // -------------------------------------------------
    // If the caller doesn't specify a date range, this
    // query used to scan the ENTIRE purchase history for
    // the company/member with no bound at all. On a table
    // that accumulates two rows per member per day, that
    // becomes a full scan that only gets slower over time
    // and is what's timing out (30s) below. Default to the
    // last 31 days so a normal dashboard load stays fast;
    // callers that explicitly want more can still pass an
    // explicit fromDate/toDate (e.g. a "full history" export).
    // =================================================

    const effectiveFromDate =
      fromDate && String(fromDate).trim() !== ""
        ? String(fromDate).trim()
        : null;

    const effectiveToDate =
      toDate && String(toDate).trim() !== ""
        ? String(toDate).trim()
        : null;

    let defaultWindowApplied = false;

    if (effectiveFromDate) {
      request.input(
        "FromDate",
        sql.Date,
        effectiveFromDate
      );

      where += `
        AND p.PurchaseDate >= @FromDate
      `;
    } else if (!effectiveToDate) {
      // No range given at all -> default window.
      defaultWindowApplied = true;

      request.input("FromDate", sql.Date, getDefaultFromDate());

      where += `
        AND p.PurchaseDate >= @FromDate
      `;
    }

    // =================================================
    // TO DATE
    // =================================================

    if (
      toDate &&
      String(toDate).trim() !== ""
    ) {
      request.input(
        "ToDate",
        sql.Date,
        String(toDate).trim()
      );

      where += `
        AND p.PurchaseDate <
            DATEADD(DAY, 1, @ToDate)
      `;
    }

    // =================================================
    // PURCHASE QUERY
    // =================================================

    const result = await request.query(`
      SELECT

        p.Purchasenumber AS PurchaseID,

        CONVERT(
          varchar(10),
          p.PurchaseDate,
          103
        ) AS PurchaseDate,

        CASE
          WHEN p.Shift = 'M'
            THEN 'Morning'

          WHEN p.Shift = 'E'
            THEN 'Evening'

          ELSE ISNULL(p.Shift, '-')
        END AS ShiftName,

        p.MemberCode,

        m.Number AS MemberNumber,

        m.MemberName,

        p.Test AS FatPercent,

        p.Snf AS SNFPercent,

        p.Qty AS QtyLtr,

        p.Rate,

        p.Amount,

        p.CompanyCode,

        RTRIM(
          ISNULL(c.Header1, '') +
          ISNULL(c.Header2, '')
        ) AS CompanyName

      FROM tbl_Purchase p

      LEFT JOIN tbl_Member m
        ON m.CompanyCode = p.CompanyCode

        AND CONVERT(
          varchar(100),
          m.MemberCode
        ) =
        CONVERT(
          varchar(100),
          p.MemberCode
        )

      OUTER APPLY
      (
        SELECT TOP 1
          co.Header1,
          co.Header2
        FROM tbl_Company co
        WHERE co.CompanyCode = p.CompanyCode
        ORDER BY co.EDNO
      ) c

      ${where}

      ORDER BY
        p.PurchaseDate DESC,

        CASE
          WHEN p.Shift = 'M' THEN 1
          WHEN p.Shift = 'E' THEN 2
          ELSE 3
        END,

        p.Purchasenumber DESC
    `);

    // =================================================
    // RECORDS
    // =================================================

    const records = result.recordset;

    // =================================================
    // SUMMARY
    // =================================================

    const summary = {
      count: records.length,

      totalQty: records.reduce(
        (sum, row) =>
          sum + Number(row.QtyLtr || 0),
        0
      ),

      totalAmount: records.reduce(
        (sum, row) =>
          sum + Number(row.Amount || 0),
        0
      ),

      fatSum: records.reduce(
        (sum, row) =>
          sum + Number(row.FatPercent || 0),
        0
      ),

      snfSum: records.reduce(
        (sum, row) =>
          sum + Number(row.SNFPercent || 0),
        0
      ),
    };

    // =================================================
    // RESPONSE
    // =================================================

    return res.json({
      success: true,

      role: me.UserTypeName,

      userTypeCode: me.UserTypeCode,

      isSecretary,

      fullAccess: isFullAccess,

      companyCode: me.CompanyCode,

      companyName: me.CompanyName || "",

      defaultWindowApplied,
      defaultLoopbackDays: defaultWindowApplied
        ? DEFAULT_LOOPBACK_DAYS
        : undefined,

      records,

      summary: {
        count: summary.count,

        totalQty: Number(
          summary.totalQty.toFixed(2)
        ),

        totalAmount: Number(
          summary.totalAmount.toFixed(2)
        ),

        avgFat: summary.count
          ? Number(
              (
                summary.fatSum /
                summary.count
              ).toFixed(2)
            )
          : 0,

        avgSnf: summary.count
          ? Number(
              (
                summary.snfSum /
                summary.count
              ).toFixed(2)
            )
          : 0,
      },
    });
  } catch (err) {
    console.error(
      "PURCHASE API ERROR:",
      err
    );

    return res.status(500).json({
      success: false,
      message:
        err.message ||
        "Failed to load purchase data.",
    });
  }
});

        0
      ),

      snfSum: records.reduce(
        (sum, row) =>
          sum + Number(row.SNFPercent || 0),
        0
      ),
    };

    // =================================================
    // RESPONSE
    // =================================================

    return res.json({
      success: true,

      role: me.UserTypeName,

      userTypeCode: me.UserTypeCode,

      isSecretary,

      fullAccess: isFullAccess,

      companyCode: me.CompanyCode,

      companyName: me.CompanyName || "",

      defaultWindowApplied,
      defaultLookbackDays: defaultWindowApplied
        ? DEFAULT_LOOKBACK_DAYS
        : undefined,

      records,

      summary: {
        count: summary.count,

        totalQty: Number(
          summary.totalQty.toFixed(2)
        ),

        totalAmount: Number(
          summary.totalAmount.toFixed(2)
        ),

        avgFat: summary.count
          ? Number(
              (
                summary.fatSum /
                summary.count
              ).toFixed(2)
            )
          : 0,

        avgSnf: summary.count
          ? Number(
              (
                summary.snfSum /
                summary.count
              ).toFixed(2)
            )
          : 0,
      },
    });
  } catch (err) {
    console.error(
      "PURCHASE API ERROR:",
      err
    );

    return res.status(500).json({
      success: false,
      message:
        err.message ||
        "Failed to load purchase data.",
    });
  }
});

import {
  useCallback,
  useEffect,
  useState,
} from "react";

import { api } from "../api";

// =====================================================
// FULL ACCESS ROLES
// =====================================================

const FULL_ACCESS_ROLES = [
  "admin",
  "manager",
];

// =====================================================
// NORMALIZE ROLE
// =====================================================

const normalizeRole = (value) => {
  return String(value || "")
    .trim()
    .toLowerCase()
    .replace(/\s+/g, " ");
};

// =====================================================
// EMPTY SUMMARY
// =====================================================

const EMPTY_SUMMARY = {
  count: 0,
  totalQty: 0,
  totalAmount: 0,
  avgFat: 0,
  avgSnf: 0,
};

// =====================================================
// EMPTY FILTERS
// =====================================================

const EMPTY_FILTERS = {
  companyCode: "",
  memberNumber: "",
  fromDate: "",
  toDate: "",
};

// =====================================================
// PURCHASE SCREEN
// =====================================================

export default function PurchaseScreen({
  user,
  onLogout,
}) {
  // ===================================================
  // USER / ROLE
  // ===================================================

  const role = normalizeRole(
    user?.userTypeName
  );

  const userTypeCode = Number(
    user?.userTypeCode
  );

  const isSecretary =
    userTypeCode === 2 ||
    role === "secretary" ||
    role === "secretory";

  const isFullAccess =
    !isSecretary &&
    FULL_ACCESS_ROLES.includes(role);

  const isMember =
    !isSecretary &&
    !isFullAccess;

  // ===================================================
  // STATES
  // ===================================================

  const [societies, setSocieties] =
    useState([]);

  const [
    secretaryCompanyName,
    setSecretaryCompanyName,
  ] = useState("");

  const [filters, setFilters] =
    useState(EMPTY_FILTERS);

  const [societySearch, setSocietySearch] =
    useState("");

  const [
    showSocietyOptions,
    setShowSocietyOptions,
  ] = useState(false);

  const [records, setRecords] =
    useState([]);

  // IMPORTANT:
  // Initial summary is ZERO.
  const [summary, setSummary] =
    useState(EMPTY_SUMMARY);

  // IMPORTANT:
  // Initial loading is FALSE.
  // No API request happens automatically.
  const [loading, setLoading] =
    useState(false);

  const [error, setError] =
    useState("");

  // ===================================================
  // LOAD SOCIETIES
  // ADMIN / MANAGER ONLY
  // ===================================================

  useEffect(() => {
    if (!isFullAccess) {
      return;
    }

    const loadSocieties = async () => {
      try {
        const data =
          await api.getSocieties();

        let list = [];

        if (Array.isArray(data)) {
          list = data;
        } else if (
          Array.isArray(data?.records)
        ) {
          list = data.records;
        } else if (
          Array.isArray(data?.companies)
        ) {
          list = data.companies;
        } else if (
          Array.isArray(data?.data)
        ) {
          list = data.data;
        }

        const cleanList = list
          .map((item) => ({
            CompanyCode:
              item?.CompanyCode ??
              item?.companyCode,

            CompanyName:
              item?.CompanyName ??
              item?.companyName,

            MobileNo:
              item?.MobileNo ??
              item?.mobileNo ??
              "",
          }))
          .filter(
            (item) =>
              item.CompanyCode !==
                undefined &&
              item.CompanyCode !==
                null &&
              String(
                item.CompanyCode
              ).trim() !== "" &&
              String(
                item.CompanyName || ""
              ).trim() !== ""
          )
          .filter(
            (item, index, array) =>
              index ===
              array.findIndex(
                (x) =>
                  String(
                    x.CompanyCode
                  ) ===
                  String(
                    item.CompanyCode
                  )
              )
          );

        setSocieties(cleanList);
      } catch (err) {
        console.error(
          "SOCIETY ERROR:",
          err
        );

        setSocieties([]);
      }
    };

    loadSocieties();
  }, [isFullAccess]);

  // ===================================================
  // LOAD SECRETARY SOCIETY NAME
  //
  // This only loads the company/society NAME.
  // It does NOT load purchase records.
  // ===================================================

  useEffect(() => {
    if (!isSecretary) {
      return;
    }

    const loadSecretarySociety =
      async () => {
        try {
          // First use company name from login
          const loginCompanyName =
            user?.companyName ??
            user?.CompanyName ??
            "";

          if (
            String(
              loginCompanyName
            ).trim() !== ""
          ) {
            setSecretaryCompanyName(
              String(
                loginCompanyName
              ).trim()
            );

            return;
          }

          // Otherwise find company using code
          const companyCode =
            user?.companyCode ??
            user?.CompanyCode;

          if (
            companyCode ===
              undefined ||
            companyCode ===
              null
          ) {
            return;
          }

          const data =
            await api.getSocieties();

          let list = [];

          if (Array.isArray(data)) {
            list = data;
          } else if (
            Array.isArray(data?.records)
          ) {
            list = data.records;
          } else if (
            Array.isArray(data?.companies)
          ) {
            list = data.companies;
          } else if (
            Array.isArray(data?.data)
          ) {
            list = data.data;
          }

          const company =
            list.find(
              (item) =>
                String(
                  item?.CompanyCode ??
                    item?.companyCode
                ) ===
                String(
                  companyCode
                )
            );

          if (company) {
            setSecretaryCompanyName(
              String(
                company?.CompanyName ??
                  company?.companyName ??
                  ""
              ).trim()
            );
          }
        } catch (err) {
          console.error(
            "SECRETARY SOCIETY ERROR:",
            err
          );
        }
      };

    loadSecretarySociety();
  }, [
    isSecretary,
    user?.companyCode,
    user?.CompanyCode,
    user?.companyName,
    user?.CompanyName,
  ]);

  // ===================================================
  // DATE FORMAT
  // ===================================================

  const formatDate = (value) => {
    if (!value) {
      return "-";
    }

    const text =
      String(value).trim();

    if (
      /^\d{2}\/\d{2}\/\d{4}$/.test(
        text
      )
    ) {
      return text;
    }

    const date =
      new Date(value);

    if (
      Number.isNaN(
        date.getTime()
      )
    ) {
      return text;
    }

    const day =
      String(
        date.getUTCDate()
      ).padStart(2, "0");

    const month =
      String(
        date.getUTCMonth() + 1
      ).padStart(2, "0");

    const year =
      date.getUTCFullYear();

    return `${day}/${month}/${year}`;
  };

  // ===================================================
  // LOAD PURCHASES
  //
  // IMPORTANT:
  // This function runs ONLY when View is clicked.
  // ===================================================

  const loadPurchases =
    useCallback(async () => {
      // -----------------------------------------------
      // USER CHECK
      // -----------------------------------------------

      if (!user?.userCode) {
        setError(
          "User information is missing. Please login again."
        );

        setRecords([]);
        setSummary(EMPTY_SUMMARY);

        return;
      }

      try {
        setLoading(true);
        setError("");

        // ---------------------------------------------
        // BASE PARAMS
        // ---------------------------------------------

        const params = {
          userCode: user.userCode,
        };

        // ---------------------------------------------
        // ADMIN / MANAGER
        // ---------------------------------------------

        if (isFullAccess) {
          if (
            filters.companyCode &&
            String(
              filters.companyCode
            ).trim() !== ""
          ) {
            params.companyCode =
              String(
                filters.companyCode
              ).trim();
          }

          if (
            filters.memberNumber &&
            String(
              filters.memberNumber
            ).trim() !== ""
          ) {
            params.memberNumber =
              String(
                filters.memberNumber
              ).trim();
          }
        }

        // ---------------------------------------------
        // SECRETARY
        // ---------------------------------------------

        if (isSecretary) {
          if (
            filters.memberNumber &&
            String(
              filters.memberNumber
            ).trim() !== ""
          ) {
            params.memberNumber =
              String(
                filters.memberNumber
              ).trim();
          }
        }

        // ---------------------------------------------
        // MEMBER
        //
        // DO NOT SEND MEMBER NUMBER.
        //
        // Backend gets logged-in member
        // using userCode.
        // ---------------------------------------------

        if (isMember) {
          console.log(
            "MEMBER PURCHASE REQUEST:",
            params
          );
        }

        // ---------------------------------------------
        // FROM DATE
        // ---------------------------------------------

        if (
          filters.fromDate &&
          String(
            filters.fromDate
          ).trim() !== ""
        ) {
          params.fromDate =
            String(
              filters.fromDate
            ).trim();
        }

        // ---------------------------------------------
        // TO DATE
        // ---------------------------------------------

        if (
          filters.toDate &&
          String(
            filters.toDate
          ).trim() !== ""
        ) {
          params.toDate =
            String(
              filters.toDate
            ).trim();
        }

        // ---------------------------------------------
        // DEBUG
        // ---------------------------------------------

        console.log(
          "======================================"
        );

        console.log(
          "PURCHASE REQUEST:",
          params
        );

        console.log(
          "======================================"
        );

        // ---------------------------------------------
        // API CALL
        // ---------------------------------------------

        const data =
          await api.getPurchases(
            params
          );

        console.log(
          "PURCHASE RESPONSE:",
          data
        );

        // ---------------------------------------------
        // RECORDS
        // ---------------------------------------------

        const purchaseRecords =
          Array.isArray(
            data?.records
          )
            ? data.records
            : [];

        setRecords(
          purchaseRecords
        );

        // ---------------------------------------------
        // SUMMARY
        // ---------------------------------------------

        setSummary({
          count: Number(
            data?.summary?.count ??
              purchaseRecords.length ??
              0
          ),

          totalQty: Number(
            data?.summary?.totalQty ??
              0
          ),

          totalAmount: Number(
            data?.summary
              ?.totalAmount ??
              0
          ),

          avgFat: Number(
            data?.summary?.avgFat ??
              0
          ),

          avgSnf: Number(
            data?.summary?.avgSnf ??
              data?.summary?.avgSNF ??
              0
          ),
        });

        // ---------------------------------------------
        // SECRETARY COMPANY NAME
        // ---------------------------------------------

        if (
          isSecretary &&
          data?.companyName
        ) {
          setSecretaryCompanyName(
            String(
              data.companyName
            ).trim()
          );
        }
      } catch (err) {
        console.error(
          "PURCHASE ERROR:",
          err
        );

        setError(
          err?.message ||
            "Unable to load purchase data."
        );

        setRecords([]);

        setSummary(
          EMPTY_SUMMARY
        );
      } finally {
        setLoading(false);
      }
    }, [
      user?.userCode,
      isFullAccess,
      isSecretary,
      isMember,
      filters.companyCode,
      filters.memberNumber,
      filters.fromDate,
      filters.toDate,
    ]);

  // ===================================================
  // IMPORTANT
  //
  // NO AUTO PURCHASE LOAD HERE.
  //
  // Do NOT add:
  //
  // useEffect(() => {
  //   loadPurchases();
  // }, [loadPurchases]);
  //
  // Purchase data loads only when View is clicked.
  // ===================================================

  // ===================================================
  // SOCIETY SEARCH
  // ===================================================

  const filteredSocieties =
    societySearch.trim() === ""
      ? societies
      : societies.filter(
          (society) => {
            const query =
              societySearch
                .trim()
                .toLowerCase();

            const name =
              String(
                society.CompanyName ||
                  ""
              ).toLowerCase();

            const mobile =
              String(
                society.MobileNo ||
                  ""
              ).toLowerCase();

            return (
              name.includes(query) ||
              mobile.includes(query)
            );
          }
        );

  // ===================================================
  // SELECT SOCIETY
  // ===================================================

  const handleSocietySelect =
    (society) => {
      setFilters(
        (prev) => ({
          ...prev,

          companyCode:
            society.CompanyCode,
        })
      );

      setSocietySearch(
        society.CompanyName
      );

      setShowSocietyOptions(
        false
      );
    };

  // ===================================================
  // SOCIETY TEXT CHANGE
  // ===================================================

  const handleSocietyTextChange =
    (e) => {
      const value =
        e.target.value;

      setSocietySearch(
        value
      );

      setShowSocietyOptions(
        true
      );

      // If society text is cleared,
      // clear company code too.
      if (
        value.trim() === ""
      ) {
        setFilters(
          (prev) => ({
            ...prev,
            companyCode: "",
          })
        );
      }
    };

  // ===================================================
  // FILTER CHANGE
  // ===================================================

  const handleFilterChange =
    (e) => {
      const {
        name,
        value,
      } = e.target;

      setFilters(
        (prev) => ({
          ...prev,
          [name]: value,
        })
      );
    };

  // ===================================================
  // CLEAR
  //
  // IMPORTANT:
  // Clear does NOT call API.
  // Everything becomes ZERO / EMPTY.
  // ===================================================

  const clearFilters = () => {
    setFilters(
      EMPTY_FILTERS
    );

    setSocietySearch("");

    setShowSocietyOptions(
      false
    );

    setRecords([]);

    setSummary(
      EMPTY_SUMMARY
    );

    setError("");

    setLoading(false);
  };

  // ===================================================
  // RECORD HELPERS
  // ===================================================

  const getPurchaseDate =
    (record) =>
      formatDate(
        record?.PurchaseDate ??
          record?.purchaseDate ??
          record?.Date ??
          record?.date
      );

  const getShift =
    (record) =>
      record?.ShiftName ??
      record?.ShiftN ??
      record?.Shift ??
      "-";

  const getMemberNumber =
    (record) =>
      record?.MemberNumber ??
      record?.MemberNo ??
      record?.Number ??
      record?.MemberCode ??
      "-";

  const getQuantity =
    (record) =>
      Number(
        record?.QtyLtr ??
          record?.Qty ??
          record?.Quantity ??
          0
      );

  const getFat =
    (record) =>
      Number(
        record?.FatPercent ??
          record?.Test ??
          record?.Fat ??
          0
      );

  const getSnf =
    (record) =>
      Number(
        record?.SNFPercent ??
          record?.Snf ??
          record?.SNF ??
          record?.snf ??
          0
      );

  const getRate =
    (record) =>
      Number(
        record?.Rate ??
          record?.rate ??
          0
      );

  const getAmount =
    (record) =>
      Number(
        record?.Amount ??
          record?.amount ??
          0
      );

  const getPurchaseId =
    (record, index) =>
      record?.PurchaseID ??
      record?.PurchaseId ??
      record?.Purchasenumber ??
      `purchase-${index}`;

  // ===================================================
  // HEADER
  // ===================================================

  const displayCompanyName =
    isSecretary
      ? secretaryCompanyName
      : user?.companyName ??
        user?.CompanyName ??
        "";

  const displayRole =
    isSecretary
      ? "SECRETORY"
      : String(
          user?.userTypeName ||
            role ||
            ""
        ).toUpperCase();

  // ===================================================
  // RENDER
  // ===================================================

  return (
    <div className="purchase-page">

      {/* =================================================
          HEADER
      ================================================= */}

      <div className="dashboard-header">

        <div>

          <h1
            className="auth-title"
            style={{
              fontSize: "26px",
              marginBottom: "8px",
            }}
          >
            Purchase Data
          </h1>

          <p
            className="auth-subtitle"
            style={{
              marginBottom: 0,
              lineHeight: "1.8",
            }}
          >
            {/* SOCIETY */}
            <strong>
              {displayCompanyName ||
                "Loading society..."}
            </strong>

            {!isMember && (
              <>
                {" · "}

                {displayRole}
              </>
            )}

            {/* MEMBER INFORMATION */}
            {isMember && (
              <div
                style={{
                  marginTop: "8px",
                  fontSize: "15px",
                  color: "#176b3a",
                }}
              >
                <div>
                  <strong>
                    Member Name:{" "}
                  </strong>

                  {user?.memberName ||
                    user?.MemberName ||
                    user?.name ||
                    user?.userName ||
                    "-"}
                </div>

                <div>
                  <strong>
                    Member No:{" "}
                  </strong>

                  {user?.memberNumber ||
                    user?.MemberNumber ||
                    user?.memberNo ||
                    user?.MemberNo ||
                    "-"}
                </div>
              </div>
            )}
          </p>
        </div>

        {onLogout && (
          <button
            type="button"
            className="btn-primary btn-logout"
            onClick={onLogout}
          >
            Log out
          </button>
        )}
      </div>

      {/* =================================================
          SUMMARY
          INITIAL VALUE = ZERO
      ================================================= */}

      {!isMember && (
        <div
          style={{
            display: "flex",
            gap: "45px",
            flexWrap: "wrap",
            marginBottom: "25px",
            padding: "5px 0",
          }}
        >

          {/* ENTRIES */}

          <div>
            <span
              style={{
                display: "block",
                fontSize: "13px",
                fontWeight: 600,
                color: "#666",
                marginBottom: "5px",
              }}
            >
              Entries
            </span>

            <strong
              style={{
                display: "block",
                fontSize: "22px",
                color: "#176b3a",
              }}
            >
              {loading
                ? "…"
                : summary.count}
            </strong>
          </div>

          {/* TOTAL QTY */}

          <div>
            <span
              style={{
                display: "block",
                fontSize: "13px",
                fontWeight: 600,
                color: "#666",
                marginBottom: "5px",
              }}
            >
              Total Qty (Ltr)
            </span>

            <strong
              style={{
                display: "block",
                fontSize: "22px",
                color: "#176b3a",
              }}
            >
              {loading
                ? "…"
                : Number(
                    summary.totalQty ||
                      0
                  ).toFixed(2)}
            </strong>
          </div>

          {/* TOTAL AMOUNT */}

          <div>
            <span
              style={{
                display: "block",
                fontSize: "13px",
                fontWeight: 600,
                color: "#666",
                marginBottom: "5px",
              }}
            >
              Total Amount
            </span>

            <strong
              style={{
                display: "block",
                fontSize: "22px",
                color: "#176b3a",
              }}
            >
              {loading
                ? "…"
                : Number(
                    summary.totalAmount ||
                      0
                  ).toFixed(2)}
            </strong>
          </div>

          {/* AVG FAT */}

          <div>
            <span
              style={{
                display: "block",
                fontSize: "13px",
                fontWeight: 600,
                color: "#666",
                marginBottom: "5px",
              }}
            >
              Avg Fat %
            </span>

            <strong
              style={{
                display: "block",
                fontSize: "22px",
                color: "#176b3a",
              }}
            >
              {loading
                ? "…"
                : Number(
                    summary.avgFat ||
                      0
                  ).toFixed(1)}
            </strong>
          </div>

          {/* AVG SNF */}

          <div>
            <span
              style={{
                display: "block",
                fontSize: "13px",
                fontWeight: 600,
                color: "#666",
                marginBottom: "5px",
              }}
            >
              Avg SNF %
            </span>

            <strong
              style={{
                display: "block",
                fontSize: "22px",
                color: "#176b3a",
              }}
            >
              {loading
                ? "…"
                : Number(
                    summary.avgSnf ||
                      0
                  ).toFixed(1)}
            </strong>
          </div>

        </div>
      )}

      {/* =================================================
          FILTER BAR
      ================================================= */}

      <div
        className="filter-bar"
        style={{
          display: "grid",

          gridTemplateColumns:
            isFullAccess
              ? "1.5fr 1fr 1fr auto"
              : "1fr 1fr auto",

          gap: "14px",

          alignItems: "end",
        }}
      >

        {/* =================================================
            ADMIN / MANAGER SOCIETY
        ================================================= */}

        {isFullAccess && (
          <label
            className="field-label filter-field"
            style={{
              position: "relative",
            }}
          >
            <span>
              Society Name
            </span>

            <input
              type="text"
              value={
                societySearch
              }
              onChange={
                handleSocietyTextChange
              }
              onFocus={() =>
                setShowSocietyOptions(
                  true
                )
              }
              className="field-input"
              placeholder="Type society name"
              autoComplete="off"
            />

            {showSocietyOptions && (
              <div className="society-suggestions">

                {filteredSocieties.length ===
                0 ? (
                  <div
                    style={{
                      padding:
                        "10px 12px",
                      color: "#777",
                    }}
                  >
                    No societies found
                  </div>
                ) : (
                  filteredSocieties.map(
                    (
                      society,
                      index
                    ) => (
                      <div
                        key={
                          society.CompanyCode ??
                          `society-${index}`
                        }
                        onMouseDown={() =>
                          handleSocietySelect(
                            society
                          )
                        }
                        style={{
                          padding:
                            "10px 12px",
                          cursor:
                            "pointer",
                        }}
                      >
                        {
                          society.CompanyName
                        }
                      </div>
                    )
                  )
                )}

              </div>
            )}
          </label>
        )}

        {/* =================================================
            SECRETARY MEMBER NUMBER
        ================================================= */}

        {isSecretary && (
          <label className="field-label filter-field">

            <span>
              Member Number
            </span>

            <input
              type="text"
              name="memberNumber"
              value={
                filters.memberNumber
              }
              onChange={
                handleFilterChange
              }
              className="field-input"
              placeholder="Enter member number"
            />

          </label>
        )}

        {/* =================================================
            FROM DATE
        ================================================= */}

        <label className="field-label filter-field">

          <span>
            From Date
          </span>

          <input
            type="date"
            name="fromDate"
            value={
              filters.fromDate
            }
            onChange={
              handleFilterChange
            }
            className="field-input"
          />

        </label>

        {/* =================================================
            TO DATE
        ================================================= */}

        <label className="field-label filter-field">

          <span>
            To Date
          </span>

          <input
            type="date"
            name="toDate"
            value={
              filters.toDate
            }
            onChange={
              handleFilterChange
            }
            className="field-input"
          />

        </label>

        {/* =================================================
            BUTTONS
        ================================================= */}

        <div
          className="filter-actions"
          style={{
            display: "flex",
            gap: "8px",
          }}
        >

          {/* VIEW */}

          <button
            type="button"
            className="btn-primary"
            onClick={
              loadPurchases
            }
            disabled={loading}
          >
            {loading
              ? "Loading..."
              : "View"}
          </button>

          {/* CLEAR */}

          <button
            type="button"
            className="btn-secondary"
            onClick={
              clearFilters
            }
            disabled={loading}
          >
            Clear
          </button>

        </div>

      </div>

      {/* =================================================
          ERROR
      ================================================= */}

      {error && (
        <div
          className="form-error"
          style={{
            marginBottom:
              "12px",
          }}
        >
          {error}
        </div>
      )}

      {/* =================================================
          TABLE
      ================================================= */}

      <div className="table-wrap">

        <table className="data-table">

          <thead>

            <tr>

              <th>
                Date
              </th>

              <th>
                Shift
              </th>

              <th>
                Member No
              </th>

              <th className="text-right">
                Qty (Ltr)
              </th>

              <th className="text-right">
                Fat %
              </th>

              <th className="text-right">
                SNF %
              </th>

              <th className="text-right">
                Rate
              </th>

              <th className="text-right">
                Amount
              </th>

            </tr>

          </thead>

          <tbody>

            {/* =========================================
                LOADING
            ========================================= */}

            {loading ? (
              <tr>

                <td
                  colSpan={8}
                  className="table-empty"
                >
                  Loading...
                </td>

              </tr>

            ) : records.length ===
              0 ? (

              /* =======================================
                 INITIAL / EMPTY
              ======================================= */

              <tr>

                <td
                  colSpan={8}
                  className="table-empty"
                >
                  No purchase records found.
                </td>

              </tr>

            ) : (

              /* =======================================
                 RECORDS
              ======================================= */

              records.map(
                (
                  record,
                  index
                ) => (

                  <tr
                    key={getPurchaseId(
                      record,
                      index
                    )}
                  >

                    <td>
                      {getPurchaseDate(
                        record
                      )}
                    </td>

                    <td>
                      {getShift(
                        record
                      )}
                    </td>

                    <td>
                      {getMemberNumber(
                        record
                      )}
                    </td>

                    <td className="text-right">
                      {getQuantity(
                        record
                      ).toFixed(1)}
                    </td>

                    <td className="text-right">
                      {getFat(
                        record
                      ).toFixed(1)}
                    </td>

                    <td className="text-right">
                      {getSnf(
                        record
                      ).toFixed(1)}
                    </td>

                    <td className="text-right">
                      {getRate(
                        record
                      ).toFixed(2)}
                    </td>

                    <td className="text-right">
                      {getAmount(
                        record
                      ).toFixed(2)}
                    </td>

                  </tr>

                )
              )
            )}

          </tbody>

        </table>

      </div>

    </div>
  );
}

const express = require("express");
const router = express.Router();
const sql = require("mssql");

const { getPool } = require("../db");

// =====================================================
// FULL ACCESS ROLES
// =====================================================

const FULL_ACCESS_ROLES = [
  "admin",
  "manager",
];

// =====================================================
// NORMALIZE ROLE
// =====================================================

const normalizeRole = (value) => {
  return String(value || "")
    .trim()
    .toLowerCase()
    .replace(/\s+/g, " ");
};

// =====================================================
// DEFAULT DATE WINDOW
// =====================================================

const DEFAULT_LOOKBACK_DAYS = 31;

function getDefaultFromDate() {
  const d = new Date();

  d.setDate(
    d.getDate() - DEFAULT_LOOKBACK_DAYS
  );

  return d.toISOString().slice(0, 10);
}

// =====================================================
// GET REGISTERED SOCIETIES
//
// GET /api/purchase-companies
//
// Secretary/Admin/Manager can use this dropdown.
// =====================================================

router.get("/purchase-companies", async (req, res) => {
  try {
    console.log("======================================");
    console.log("GET /api/purchase-companies");
    console.log("======================================");

    const pool = await getPool();

    const result = await pool.request().query(`
      SELECT
        CompanyCode,
        EDNO,
        Header1,
        Header2,
        MobileNo
      FROM tbl_Company
      WHERE
        LEN(ISNULL(MobileNo, '')) = 10
        AND EDNO > 0
      ORDER BY
        CompanyCode
    `);

    const companies = result.recordset.map((row) => ({
      CompanyCode: row.CompanyCode,

      EDNO: row.EDNO,

      CompanyName:
        `${row.Header1 || ""} ${row.Header2 || ""}`.trim(),

      MobileNo: row.MobileNo,
    }));

    console.log(
      "REGISTERED SOCIETIES:",
      companies.length
    );

    return res.status(200).json({
      success: true,
      records: companies,
    });

  } catch (err) {
    console.error(
      "GET /api/purchase-companies ERROR:",
      err
    );

    return res.status(500).json({
      success: false,
      message:
        err.message ||
        "Failed to load societies.",
    });
  }
});

// =====================================================
// GET PURCHASES
//
// GET /api/purchases
//
// Query:
// userCode
// companyCode
// memberNumber
// fromDate
// toDate
// =====================================================

router.get("/purchases", async (req, res) => {

  try {

    // =================================================
    // REQUEST PARAMETERS
    // =================================================

    const {
      userCode,
      companyCode,
      memberNumber,
      fromDate,
      toDate,
    } = req.query;

    console.log("======================================");
    console.log("PURCHASE API REQUEST");
    console.log("Query:", req.query);
    console.log("======================================");

    // =================================================
    // VALIDATE USER CODE
    // =================================================

    if (
      userCode === undefined ||
      userCode === null ||
      String(userCode).trim() === ""
    ) {
      return res.status(400).json({
        success: false,
        message: "userCode is required.",
      });
    }

    const numericUserCode = Number(userCode);

    if (
      !Number.isInteger(numericUserCode) ||
      numericUserCode <= 0
    ) {
      return res.status(400).json({
        success: false,
        message: "Invalid userCode.",
      });
    }

    // =================================================
    // DATABASE
    // =================================================

    const pool = await getPool();

    // =================================================
    // GET LOGGED-IN USER
    // =================================================

    const userResult =
      await pool
        .request()
        .input(
          "UserCode",
          sql.Int,
          numericUserCode
        )
        .query(`
          SELECT

            u.UserCode,

            u.UserTypeCode,

            u.CompanyCode,

            u.MemberNumber,

            u.RegisteredMobileNumber,

            u.UserName,

            t.UserTypeName,

            RTRIM(
              ISNULL(c.Header1, '') +
              ISNULL(c.Header2, '')
            ) AS CompanyName

          FROM tbl_User u

          INNER JOIN tbl_UserType t
            ON u.UserTypeCode =
               t.UserTypeCode

          OUTER APPLY
          (
            SELECT TOP 1
              co.Header1,
              co.Header2
            FROM tbl_Company co
            WHERE
              co.CompanyCode =
              u.CompanyCode
            ORDER BY
              co.EDNO
          ) c

          WHERE
            u.UserCode =
            @UserCode
        `);

    // =================================================
    // USER NOT FOUND
    // =================================================

    if (
      userResult.recordset.length === 0
    ) {
      return res.status(404).json({
        success: false,
        message: "User not found.",
      });
    }

    const me =
      userResult.recordset[0];

    // =================================================
    // ROLE
    // =================================================

    const role =
      normalizeRole(
        me.UserTypeName
      );

    const userTypeCode =
      Number(me.UserTypeCode);

    // =================================================
    // SECRETARY
    //
    // UserTypeCode 2 = Secretary
    // =================================================

    const isSecretary =
      userTypeCode === 2 ||
      role === "secretary" ||
      role === "secretory";

    // =================================================
    // ADMIN
    // =================================================

    const isAdmin =
      role === "admin";

    // =================================================
    // MANAGER
    // =================================================

    const isManager =
      role === "manager";

    // =================================================
    // FULL ACCESS
    //
    // Secretary
    // Admin
    // Manager
    // =================================================

    const isFullAccess =
      isSecretary ||
      isAdmin ||
      isManager ||
      FULL_ACCESS_ROLES.includes(role);

    // =================================================
    // ACCESS LOG
    // =================================================

    console.log("======================================");
    console.log("ACCESS CHECK");
    console.log("UserCode:", me.UserCode);
    console.log("UserTypeCode:", userTypeCode);
    console.log("UserTypeName:", me.UserTypeName);
    console.log("Normalized Role:", role);
    console.log("Registered CompanyCode:", me.CompanyCode);
    console.log("Registered MemberNumber:", me.MemberNumber);
    console.log(
      "Registered Mobile:",
      me.RegisteredMobileNumber
    );
    console.log("Requested CompanyCode:", companyCode);
    console.log("Requested MemberNumber:", memberNumber);
    console.log("isSecretary:", isSecretary);
    console.log("isAdmin:", isAdmin);
    console.log("isManager:", isManager);
    console.log("isFullAccess:", isFullAccess);
    console.log("======================================");

    // =================================================
    // BUILD WHERE CLAUSE
    // =================================================

    const request =
      pool.request();

    let where =
      "WHERE 1 = 1";

    // =================================================
    // SECRETARY / ADMIN / MANAGER
    //
    // Can filter society.
    //
    // Can optionally filter member.
    //
    // Empty member = ALL MEMBERS.
    // =================================================

    if (isFullAccess) {

      console.log(
        "FULL ACCESS USER"
      );

      // =================================================
      // SOCIETY FILTER
      // =================================================

      if (
        companyCode !== undefined &&
        companyCode !== null &&
        String(companyCode).trim() !== ""
      ) {

        const selectedCompanyCode =
          Number(companyCode);

        if (
          !Number.isInteger(
            selectedCompanyCode
          ) ||
          selectedCompanyCode <= 0
        ) {
          return res.status(400).json({
            success: false,
            message:
              "Invalid companyCode.",
          });
        }

        request.input(
          "CompanyCode",
          sql.Int,
          selectedCompanyCode
        );

        where += `
          AND tbl_Purchase.CompanyCode =
              @CompanyCode
        `;

        console.log(
          "SOCIETY FILTER:",
          selectedCompanyCode
        );

      } else {

        console.log(
          "NO SOCIETY SELECTED - ALL SOCIETIES"
        );
      }

      // =================================================
      // MEMBER FILTER
      // =================================================

      if (
        memberNumber !== undefined &&
        memberNumber !== null &&
        String(memberNumber).trim() !== ""
      ) {

        const selectedMemberNumber =
          String(memberNumber).trim();

        request.input(
          "MemberNumber",
          sql.VarChar(50),
          selectedMemberNumber
        );

        where += `
          AND CONVERT(
                varchar(50),
                tbl_Purchase.Number
              ) =
              @MemberNumber
        `;

        console.log(
          "MEMBER FILTER:",
          selectedMemberNumber
        );

      } else {

        console.log(
          "NO MEMBER FILTER - ALL MEMBERS"
        );
      }

    }

    // =================================================
    // NORMAL MEMBER
    //
    // FORCE registered society.
    //
    // FORCE registered member number.
    //
    // Ignore companyCode/memberNumber from frontend.
    // =================================================

    else {

      console.log(
        "NORMAL MEMBER ACCESS"
      );

      // =================================================
      // REGISTERED SOCIETY REQUIRED
      // =================================================

      if (
        me.CompanyCode === null ||
        me.CompanyCode === undefined ||
        String(me.CompanyCode).trim() === ""
      ) {

        return res.status(400).json({
          success: false,
          message:
            "User has no registered society.",
        });
      }

      const registeredCompanyCode =
        Number(me.CompanyCode);

      if (
        !Number.isInteger(
          registeredCompanyCode
        ) ||
        registeredCompanyCode <= 0
      ) {

        return res.status(400).json({
          success: false,
          message:
            "Invalid registered society.",
        });
      }

      // =================================================
      // FORCE USER SOCIETY
      // =================================================

      request.input(
        "UserCompanyCode",
        sql.Int,
        registeredCompanyCode
      );

      where += `
        AND tbl_Purchase.CompanyCode =
            @UserCompanyCode
      `;

      // =================================================
      // REGISTERED MEMBER REQUIRED
      // =================================================

      if (
        me.MemberNumber === null ||
        me.MemberNumber === undefined ||
        String(me.MemberNumber).trim() === ""
      ) {

        return res.status(400).json({
          success: false,
          message:
            "User has no registered member number.",
        });
      }

      const registeredMemberNumber =
        String(
          me.MemberNumber
        ).trim();

      // =================================================
      // FORCE USER MEMBER
      // =================================================

      request.input(
        "UserMemberNumber",
        sql.VarChar(50),
        registeredMemberNumber
      );

      where += `
        AND CONVERT(
              varchar(50),
              tbl_Purchase.Number
            ) =
            @UserMemberNumber
      `;

      console.log(
        "FORCED MEMBER COMPANY:",
        registeredCompanyCode
      );

      console.log(
        "FORCED MEMBER NUMBER:",
        registeredMemberNumber
      );
    }

    // =================================================
    // DATE PARAMETERS
    // =================================================

    const effectiveFromDate =
      fromDate &&
      String(fromDate).trim() !== ""
        ? String(fromDate).trim()
        : null;

    const effectiveToDate =
      toDate &&
      String(toDate).trim() !== ""
        ? String(toDate).trim()
        : null;

    let defaultWindowApplied =
      false;

    // =================================================
    // FROM DATE
    // =================================================

    if (effectiveFromDate) {

      request.input(
        "FromDate",
        sql.Date,
        effectiveFromDate
      );

      where += `
        AND tbl_Purchase.PurchaseDate >=
            @FromDate
      `;
    }

    // =================================================
    // DEFAULT 31 DAYS
    // =================================================

    else if (!effectiveToDate) {

      defaultWindowApplied =
        true;

      request.input(
        "FromDate",
        sql.Date,
        getDefaultFromDate()
      );

      where += `
        AND tbl_Purchase.PurchaseDate >=
            @FromDate
      `;
    }

    // =================================================
    // TO DATE
    // =================================================

    if (effectiveToDate) {

      request.input(
        "ToDate",
        sql.Date,
        effectiveToDate
      );

      where += `
        AND tbl_Purchase.PurchaseDate <
            DATEADD(
              DAY,
              1,
              @ToDate
            )
      `;
    }

    // =================================================
    // FINAL WHERE
    // =================================================

    console.log("======================================");
    console.log("FINAL PURCHASE WHERE");
    console.log(where);
    console.log("======================================");

    // =================================================
    // PURCHASE QUERY
    // =================================================

    const result =
      await request.query(`

        SELECT

          tbl_Purchase.PurchaseNumber
            AS PurchaseID,

          tbl_Purchase.PurchaseDate
            AS PurchaseDate,

          CASE
            WHEN tbl_Purchase.Shift = 'M'
              THEN 'Morning'

            WHEN tbl_Purchase.Shift = 'E'
              THEN 'Evening'

            ELSE ISNULL(
              tbl_Purchase.Shift,
              '-'
            )
          END AS ShiftName,

          tbl_Purchase.MemberCode,

          tbl_Purchase.Number
            AS MemberNumber,

          tbl_Member.MemberName,

          -- =========================================
          -- MEMBER REGISTERED MOBILE
          -- =========================================

          tbl_Member.MobileNo
            AS MemberMobileNo,

          tbl_Purchase.Test
            AS FatPercent,

          tbl_Purchase.SNF
            AS SNFPercent,

          tbl_Purchase.Qty
            AS QtyLtr,

          tbl_Purchase.Rate,

          tbl_Purchase.Amount,

          tbl_Purchase.CompanyCode

        FROM tbl_Purchase

        LEFT OUTER JOIN tbl_Member

          ON tbl_Purchase.MemberCode =
             tbl_Member.MemberCode

          AND tbl_Purchase.CompanyCode =
              tbl_Member.CompanyCode

        ${where}

        ORDER BY

          tbl_Purchase.PurchaseDate DESC,

          CASE
            WHEN tbl_Purchase.Shift = 'M'
              THEN 1

            WHEN tbl_Purchase.Shift = 'E'
              THEN 2

            ELSE 3
          END,

          tbl_Purchase.PurchaseNumber DESC

      `);

    // =================================================
    // RECORDS
    // =================================================

    const records =
      result.recordset || [];

    // =================================================
    // SUMMARY
    // =================================================

    const totalQty =
      records.reduce(
        (sum, row) =>
          sum +
          Number(
            row.QtyLtr || 0
          ),
        0
      );

    const totalAmount =
      records.reduce(
        (sum, row) =>
          sum +
          Number(
            row.Amount || 0
          ),
        0
      );

    const fatSum =
      records.reduce(
        (sum, row) =>
          sum +
          Number(
            row.FatPercent || 0
          ),
        0
      );

    const snfSum =
      records.reduce(
        (sum, row) =>
          sum +
          Number(
            row.SNFPercent || 0
          ),
        0
      );

    // =================================================
    // SELECTED SOCIETY
    // =================================================

    let selectedCompany = null;

    if (
      companyCode !== undefined &&
      companyCode !== null &&
      String(companyCode).trim() !== ""
    ) {

      const selectedCompanyCode =
        Number(companyCode);

      if (
        Number.isInteger(
          selectedCompanyCode
        ) &&
        selectedCompanyCode > 0
      ) {

        const selectedCompanyResult =
          await pool
            .request()
            .input(
              "SelectedCompanyCode",
              sql.Int,
              selectedCompanyCode
            )
            .query(`
              SELECT TOP 1

                CompanyCode,

                EDNO,

                Header1,

                Header2,

                MobileNo

              FROM tbl_Company

              WHERE
                CompanyCode =
                @SelectedCompanyCode

              ORDER BY
                EDNO
            `);

        if (
          selectedCompanyResult.recordset.length > 0
        ) {

          const row =
            selectedCompanyResult.recordset[0];

          selectedCompany = {

            CompanyCode:
              row.CompanyCode,

            EDNO:
              row.EDNO,

            CompanyName:
              `${row.Header1 || ""} ${row.Header2 || ""}`
                .trim(),

            MobileNo:
              row.MobileNo,
          };
        }
      }
    }

    // =================================================
    // RESPONSE
    // =================================================

    console.log("======================================");
    console.log("PURCHASE API SUCCESS");
    console.log("Records:", records.length);
    console.log(
      "Selected Company:",
      companyCode || "ALL"
    );
    console.log(
      "Selected Member:",
      memberNumber || "ALL"
    );
    console.log(
      "Secretary:",
      isSecretary
    );
    console.log(
      "Full Access:",
      isFullAccess
    );
    console.log("======================================");

    return res.status(200).json({

      success: true,

      // =================================================
      // USER
      // =================================================

      role:
        me.UserTypeName,

      userTypeCode:
        me.UserTypeCode,

      isSecretary,

      isAdmin,

      isManager,

      fullAccess:
        isFullAccess,

      // =================================================
      // LOGGED-IN USER
      // =================================================

      userCode:
        me.UserCode,

      userName:
        me.UserName,

      registeredMobileNumber:
        me.RegisteredMobileNumber,

      userCompanyCode:
        me.CompanyCode,

      userCompanyName:
        me.CompanyName || "",

      userMemberNumber:
        me.MemberNumber || null,

      // =================================================
      // SELECTED SOCIETY
      // =================================================

      selectedCompanyCode:
        companyCode
          ? Number(companyCode)
          : null,

      selectedCompany,

      // =================================================
      // SELECTED MEMBER
      // =================================================

      selectedMemberNumber:
        memberNumber || null,

      // =================================================
      // DATE
      // =================================================

      fromDate:
        effectiveFromDate ||
        null,

      toDate:
        effectiveToDate ||
        null,

      defaultWindowApplied,

      defaultLookbackDays:
        defaultWindowApplied
          ? DEFAULT_LOOKBACK_DAYS
          : undefined,

      // =================================================
      // DATA
      // =================================================

      records,

      // =================================================
      // SUMMARY
      // =================================================

      summary: {

        count:
          records.length,

        totalQty:
          Number(
            totalQty.toFixed(2)
          ),

        totalAmount:
          Number(
            totalAmount.toFixed(2)
          ),

        avgFat:
          records.length
            ? Number(
                (
                  fatSum /
                  records.length
                ).toFixed(2)
              )
            : 0,

        avgSnf:
          records.length
            ? Number(
                (
                  snfSum /
                  records.length
                ).toFixed(2)
              )
            : 0,
      },
    });

  } catch (err) {

    // =================================================
    // ERROR
    // =================================================

    console.error(
      "======================================"
    );

    console.error(
      "PURCHASE API ERROR"
    );

    console.error(err);

    console.error(
      "======================================"
    );

    return res.status(500).json({

      success: false,

      message:
        err.message ||
        "Failed to load purchase data.",
    });
  }
});
